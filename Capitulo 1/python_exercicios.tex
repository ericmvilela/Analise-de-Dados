\section{Exercícios - Capítulo 1}

\subsection*{Exercício 1.1 -- hello world}
\begin{exercicio}{Sumário}
\begin{itemize}
    \item \textbf{Objetivo:} Complete o programa para imprimir exatamente ``Hello, world!''. Atenção à indentação.
\end{itemize}
\end{exercicio}

\subsection*{Exercício 1.2 -- compliment}
\begin{exercicio}{Sumário}
\begin{itemize}
    \item \textbf{Objetivo:} Leia o país informado pelo usuário e imprima uma frase dizendo que você ouviu que esse país é bonito.
\end{itemize}
\end{exercicio}

\subsection*{Exercício 1.3 -- multiplication}
\begin{exercicio}{Sumário}
\begin{itemize}
    \item \textbf{Objetivo:} Use um laço \texttt{for} para imprimir a tabuada do 4, de 0 a 10, no formato mostrado no exercício.
\end{itemize}
\end{exercicio}

\subsection*{Exercício 1.4 -- multiplication table}
\begin{exercicio}{Sumário}
\begin{itemize}
    \item \textbf{Objetivo:} Use dois laços \texttt{for} aninhados para imprimir a tabuada de 1 a 10. O exercício também pede alinhamento por largura fixa.
\end{itemize}
\end{exercicio}

\subsection*{Exercício 1.5 -- two dice}
\begin{exercicio}{Sumário}
\begin{itemize}
    \item \textbf{Objetivo:} Percorra, com dois \texttt{for} aninhados, todas as 36 combinações ordenadas de dois dados e imprima as que somam 5.
\end{itemize}
\end{exercicio}

\subsection*{Exercício 1.6 -- triple square}
\begin{exercicio}{Sumário}
\begin{itemize}
    \item \textbf{Objetivo:} Crie \texttt{triple(x)}, que multiplica por 3, e \texttt{square(x)}, que eleva ao quadrado. No programa principal, percorra de 1 a 10 e mostre os dois resultados; depois altere o laço para parar antes da primeira iteração em que o quadrado ultrapassar o triplo.
\end{itemize}
\end{exercicio}

\subsection*{Exercício 1.7 -- areas of shapes}
\begin{exercicio}{Sumário}
\begin{itemize}
    \item \textbf{Objetivo:} Crie um programa interativo que calcule a área de triângulo, retângulo ou círculo. Uma entrada vazia encerra o programa e uma forma desconhecida produz ``Unknown shape!''. Use o especificador \texttt{f} na impressão da área.
\end{itemize}
\end{exercicio}

\subsection*{Exercício 1.8 -- solve quadratic}
\begin{exercicio}{Sumário}
\begin{itemize}
    \item \textbf{Objetivo:} Implemente uma função que receba \(a,b,c\) e devolva as duas raízes da equação quadrática \(ax^2+bx+c=0\) como uma tupla. A fórmula é
    \[ x=\frac{-b\pm\sqrt{b^2-4ac}}{2a}. \]
\end{itemize}
\end{exercicio}

\subsection*{Exercício 1.9 -- merge}
\begin{exercicio}{Sumário}
\begin{itemize}
    \item \textbf{Objetivo:} Mescle duas listas já ordenadas em ordem crescente sem usar \texttt{sort()} nem \texttt{sorted()} dentro da função. Não modifique as listas recebidas.
\end{itemize}
\end{exercicio}

\subsection*{Exercício 1.10 -- detect ranges}
\begin{exercicio}{Sumário}
\begin{itemize}
    \item \textbf{Objetivo:} Ordene uma lista de inteiros e substitua sequências consecutivas por tuplas de intervalo. Por exemplo, 4,5,6,7,8 torna-se \texttt{(4,9)}, seguindo a convenção de limite final exclusivo usada em \texttt{range()}.
\end{itemize}
\end{exercicio}

\subsection*{Exercício 1.11 -- interleave}
\begin{exercicio}{Sumário}
\begin{itemize}
    \item \textbf{Objetivo:} Implemente \texttt{interleave()} para receber várias listas de mesmo tamanho e devolver seus elementos intercalados. Use \texttt{zip()} e o método \texttt{extend()}.
\end{itemize}
\end{exercicio}

\subsection*{Exercício 1.12 -- distinct characters}
\begin{exercicio}{Sumário}
\begin{itemize}
    \item \textbf{Objetivo:} Para cada string de uma lista, retorne em um dicionário a quantidade de caracteres distintos. Use um conjunto para armazenar temporariamente os caracteres diferentes.
\end{itemize}
\end{exercicio}

\subsection*{Exercício 1.13 -- reverse dictionary}
\begin{exercicio}{Sumário}
\begin{itemize}
    \item \textbf{Objetivo:} Dado um dicionário inglês--finlandês, crie o dicionário inverso finlandês--inglês. Como uma palavra pode ter sinônimos ou homônimos, os valores do dicionário resultante devem ser listas.
\end{itemize}
\end{exercicio}

\subsection*{Exercício 1.14 -- find matching}
\begin{exercicio}{Sumário}
\begin{itemize}
    \item \textbf{Objetivo:} Retorne os índices dos elementos de uma lista de strings que contêm uma determinada substring. Use \texttt{enumerate()}.
\end{itemize}
\end{exercicio}

\subsection*{Exercício 1.15 -- two dice comprehension}
\begin{exercicio}{Sumário}
\begin{itemize}
    \item \textbf{Objetivo:} Resolva novamente o exercício dos dois dados, mas agora use uma list comprehension e imprima cada par em uma linha.
\end{itemize}
\end{exercicio}

\subsection*{Exercício 1.16 -- transform}
\begin{exercicio}{Sumário}
\begin{itemize}
    \item \textbf{Objetivo:} Receba duas strings de números, use \texttt{split()}, \texttt{map()} e \texttt{zip()} e devolva a multiplicação posição a posição.
\end{itemize}
\end{exercicio}

\subsection*{Exercício 1.17 -- positive list}
\begin{exercicio}{Sumário}
\begin{itemize}
    \item \textbf{Objetivo:} Retorne apenas os valores positivos de uma lista, filtrando zero e números negativos com \texttt{filter()}.
\end{itemize}
\end{exercicio}

\subsection*{Exercício 1.18 -- acronyms}
\begin{exercicio}{Sumário}
\begin{itemize}
    \item \textbf{Objetivo:} Crie uma função que encontre acrônimos em um texto. Uma palavra é considerada acrônimo quando tem pelo menos dois caracteres e todos os caracteres permanecentes após a limpeza da pontuação estão em maiúsculas. Números também podem aparecer.
\end{itemize}
\end{exercicio}

\subsection*{Exercício 1.19 -- sum equation}
\begin{exercicio}{Sumário}
\begin{itemize}
    \item \textbf{Objetivo:} Receba uma lista de inteiros positivos e produza uma string com a equação da soma, por exemplo ``1 + 5 + 7 = 13''. Para lista vazia, devolva ``0 = 0''. Os espaços devem ser exatamente os indicados.
\end{itemize}
\end{exercicio}

\subsection*{Exercício 1.20 -- usemodule}
\begin{exercicio}{Sumário}
\begin{itemize}
    \item \textbf{Objetivo:} Crie o módulo \texttt{triangle.py} com duas funções: \texttt{hypotenuse()}, que calcula a hipotenusa, e \texttt{area()}, que calcula a área de um triângulo retângulo. O módulo deve ter docstrings descritivas e os atributos \texttt{\_\_version\_\_} e \texttt{\_\_author\_\_}. Chame as funções a partir de \texttt{usemodule.py}.
\end{itemize}
\end{exercicio}
